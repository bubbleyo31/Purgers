
var go=UnityEngine.GameObject.Find("GrappleLandingVerificationPlayer");
if(go==null) return "missing test player";
var g=go.GetComponent<PlayerGrapple>();
var k=go.GetComponent<Fusion.Addons.KCC.KCC>();
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic;
System.Action<string,object> set=(name,value)=>typeof(PlayerGrapple).GetProperty(name,flags).SetValue(g,value,null);
System.Func<string,bool> flag=name=>(bool)(Fusion.NetworkBool)typeof(PlayerGrapple).GetProperty(name,flags).GetValue(g,null);
System.Action<string,object> field=(name,value)=>typeof(PlayerGrapple).GetField(name,flags).SetValue(g,value);
System.Action<bool> grounded=value=>{k.FixedData.IsGrounded=value;k.RenderData.IsGrounded=value;};
System.Action tick=()=>typeof(PlayerGrapple).GetMethod("TickGrapplePhase",flags).Invoke(g,null);
System.Action<bool> begin=value=>{
    grounded(value);
    k.SetDynamicVelocity(UnityEngine.Vector3.zero);
    set("SupportInteractionTetherActive",(Fusion.NetworkBool)false);
    set("TrackingDynamicAnchor",(Fusion.NetworkBool)false);
    float height=(float)typeof(PlayerGrapple).GetField("rayOriginHeight",flags).GetValue(g);
    set("GrappleWorldPoint",k.Data.TargetPosition+new UnityEngine.Vector3(0,height*0.5f,30));
    typeof(PlayerGrapple).GetMethod("BeginAttached",flags).Invoke(g,null);
};
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>{if(!ok) throw new System.Exception("FAIL: "+name);results.Add("PASS "+name);};
field("grappleOcclusionObstacleMask",(UnityEngine.LayerMask)0);
field("debugGrapple",false);
field("enableGrappleStraightPull",true);

field("releaseNormalGrappleOnLanding",true);
begin(false);
check(!flag("HasBeenAirborneDuringAttached")&&!flag("HasAppliedAttachedPull"),"airborne attach is not armed");
grounded(true);tick();
check(g.CurrentPhase==GrapplePhase.Attached&&flag("HasAppliedAttachedPull"),"first-pull landing keeps attached and submits pull");
check(flag("HasAttemptedAttachedJump")&&k.Data.DynamicVelocity.y>0,"startup landing receives existing jump");
UnityEngine.Vector3 protectedVelocity;
check(g.TryGetProtectedAttachedDynamicVelocity(out protectedVelocity)&&protectedVelocity.y==k.Data.DynamicVelocity.y,"startup jump reaches KCC velocity protection");
k.SetDynamicVelocity(UnityEngine.Vector3.zero); tick();
check(k.Data.DynamicVelocity.y<0.01f,"grounded startup does not inject a second jump");
grounded(false);tick();
check(flag("HasBeenAirborneDuringAttached"),"airborne observation after pull arms landing");
grounded(true);tick();
check(g.CurrentPhase!=GrapplePhase.Attached,"normal later landing retracts");
check(!flag("HasAppliedAttachedPull")&&!flag("HasAttemptedAttachedJump"),"retract resets startup snapshot");
begin(true);
check(flag("HasAttemptedAttachedJump")&&!flag("HasAppliedAttachedPull"),"ground launch jumps without claiming a pull");
grounded(false);tick();
check(!flag("HasBeenAirborneDuringAttached")&&flag("HasAppliedAttachedPull"),"first pull does not use pre-pull airborne observation");
tick();check(flag("HasBeenAirborneDuringAttached"),"subsequent simulation observation arms ground launch");
field("releaseNormalGrappleOnLanding",false);
grounded(true);tick();check(g.CurrentPhase==GrapplePhase.Attached,"landing release toggle remains respected");
field("releaseNormalGrappleOnLanding",true);
begin(true);
set("AttachedStartupGroundedTimer",Fusion.TickTimer.CreateFromTicks(g.Runner,0));
check(((Fusion.TickTimer)typeof(PlayerGrapple).GetProperty("AttachedStartupGroundedTimer",flags).GetValue(g,null)).Expired(g.Runner),"test timeout is expired");
tick();check(g.CurrentPhase!=GrapplePhase.Attached,"blocked startup expires before another pull");
field("useHoldToMaintainGrappleInput",false);
begin(false);
g.Simulate(true,true,false,false,UnityEngine.Vector2.zero,1f);
check(g.CurrentPhase!=GrapplePhase.Attached,"manual toggle can cancel startup");

field("useHoldToMaintainGrappleInput",true);
begin(false);g.Simulate(false,false,false,false,UnityEngine.Vector2.zero,1f);
check(g.CurrentPhase!=GrapplePhase.Attached,"hold release cancels startup");
begin(false);g.Simulate(false,true,true,true,UnityEngine.Vector2.zero,1f);
check(g.CurrentPhase!=GrapplePhase.Attached,"aim cancels startup");
begin(false);
set("SupportInteractionTetherActive",(Fusion.NetworkBool)true);
set("GrappleHitNetworkObject",g.Object);
set("GrappleHitObjectLocalPoint",UnityEngine.Vector3.zero);
set("PreviousGrapplePoint",g.transform.position);
set("GrappleWorldPoint",g.transform.position);
set("AttachedStartupGroundedTimer",Fusion.TickTimer.CreateFromTicks(g.Runner,0));
grounded(true);tick();
check(g.CurrentPhase==GrapplePhase.Attached&&!flag("HasAppliedAttachedPull")&&!flag("HasAttemptedAttachedJump"),"support tether bypasses startup and grounded cancellation");

field("enableGrappleStraightPull",false);
field("enableGrappleLeftSwing",true);
field("enableGrappleRightSwing",false);
begin(false);grounded(true);tick();
check(g.CurrentPhase==GrapplePhase.Attached&&flag("GrappleSwingActive")&&flag("HasAppliedAttachedPull"),"left swing submits pull and survives startup landing");
grounded(false);tick();grounded(true);tick();
check(g.CurrentPhase!=GrapplePhase.Attached,"left swing later landing retracts");
field("enableGrappleLeftSwing",false);
field("enableGrappleRightSwing",true);
begin(false);grounded(true);tick();
check(g.CurrentPhase==GrapplePhase.Attached&&flag("GrappleSwingActive")&&flag("HasAppliedAttachedPull"),"right swing submits pull and survives startup landing");
return new {mode=g.Runner.GameMode.ToString(),stateAuthority=g.Object.HasStateAuthority,checks=results.ToArray(),tick=g.Runner.Tick.Raw};
